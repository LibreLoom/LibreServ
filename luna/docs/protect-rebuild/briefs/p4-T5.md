## Tables
```sql
CREATE TABLE IF NOT EXISTS private_challenges (
  hash TEXT PRIMARY KEY,         -- hex(sha256(challenge))
  account_id TEXT NOT NULL, luna_id TEXT NOT NULL,
  link_id TEXT NOT NULL DEFAULT '',   -- share link id, '' for account sessions
  expires_at INTEGER NOT NULL
);
CREATE TABLE IF NOT EXISTS private_sessions (
  token_hash TEXT PRIMARY KEY,   -- hex(sha256(token))
  account_id TEXT NOT NULL, luna_id TEXT NOT NULL, scope TEXT NOT NULL,
  link_id TEXT NOT NULL DEFAULT '',
  expires_at INTEGER NOT NULL
);
```
Postgres uses `BIGINT` where the existing Postgres schema uses it for unix times; copy the style of `share_links` in each schema.

## Go API (in private_open.go)
```go
const privateChallengeTTL = 5 * 60
const privateSessionTTL = 30 * 60
const privateOpenDomain = "private-open"
const privateScopeDomain = "private-scope"
const privateTokenMessage = "Enter your recovery phrase again to open your private items."

// lunaCanonical mirrors crypto::canonical: for each field, u32 BE len(name) ‖ name ‖ u32 BE len(value) ‖ value.
func lunaCanonical(fields ...[2][]byte) []byte
// lunaVerify checks Ed25519 over "luna-sig-v1\x00" + domain + "\x00" + msg.
func lunaVerify(pub []byte, domain string, msg, sig []byte) bool
func scopePart(userID string) string // hex(sha256(userID)[:16])

type scopeRecord struct { LunaID, Scope, UserID string; X25519Public, AuthPublic []byte; CreatedAt int64; SealedDedup []byte }
// parseScopeRecord decodes the pscopes JSON (hex fields) and verifies its signature with the Luna's signing key.
func parseScopeRecord(data []byte, lunaSigningHex string) (scopeRecord, bool)
// loadScopeRecord reads pscopes/<scope> for s (account, luna) and verifies it. ok=false if missing or bad.
func loadScopeRecord(d Deps, s readScope, scope string) (scopeRecord, bool, error)

// newPrivateChallenge stores and returns a fresh 32-byte challenge (hex).
func newPrivateChallenge(d Deps, s readScope, linkID string, now time.Time) (string, error)
// openPrivateSession checks challenge (single use, not expired, same account/luna/link) and the
// auth-key signature against the scope record's auth_public, and for a user share link that
// scopePart(record.UserID) == s.Value. A backup link (s.Kind == "backup") never opens one.
// Returns the token (hex 32 bytes) and expiry.
func openPrivateSession(d Deps, s readScope, linkID, scope, challengeHex, sigHex string, now time.Time) (token string, expiresAt int64, err error)
// privateSessionScope returns the scope a token opens, or "" if invalid/expired/other luna or link.
func privateSessionScope(d Deps, s readScope, linkID, token string, now time.Time) (string, error)

// PrivateHandler serves the HTTP side; T14 wires the routes.
type PrivateHandler struct{ Deps }
func (h PrivateHandler) Challenge(w http.ResponseWriter, r *http.Request)      // account: {lunaID} path param
func (h PrivateHandler) Open(w http.ResponseWriter, r *http.Request)           // body {scope, challenge, signature}
func (h PrivateHandler) ShareChallenge(w http.ResponseWriter, r *http.Request) // resolveShare first
func (h PrivateHandler) ShareOpen(w http.ResponseWriter, r *http.Request)
```
Rules:
- Signed `pscopes` message: `lunaCanonical(luna_id, scope, user_id, x25519_public(32), auth_public(32), created_at as 8-byte big-endian, sealed_dedup)` with names exactly `luna_id, scope, user_id, x25519_public, auth_public, created_at, sealed_dedup`; domain `private-scope`.
- Open message: `lunaCanonical(("luna_id", lunaID), ("scope", scope), ("challenge", raw32))`, domain `private-open`, key `record.AuthPublic`.
- Also reject when `record.LunaID != lunaID` or `record.Scope != scope`.
- Any failed open → 403 with `privateTokenMessage`. Bad JSON → 400 "That request could not be read. Try again.". A Luna not on the account → reuse the 404 the account read handlers use (`grep -n "StatusNotFound" cloud_read.go`).
- Delete used and expired challenges (lazy cleanup on each new challenge: `DELETE … WHERE expires_at < now`).
- Store only SHA-256 hashes of challenges and tokens.
- Use `?` placeholders like the rest of the handlers (the DB layer rewrites them for Postgres; confirm with `grep -n "Rebind\|\\$1" -r internal/database | head`).
